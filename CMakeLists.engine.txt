cmake_minimum_required(VERSION 3.16)
project(FarmEyeEngine LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# vcpkg toolchain önerisi:
#   cmake -B build-engine -DCMAKE_TOOLCHAIN_FILE=C:/vcpkg/scripts/buildsystems/vcpkg.cmake

find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs highgui dnn)

option(FARMEYE_ENABLE_GSTREAMER "GStreamer capture" ON)

set(ENGINE_SOURCES
    src/engine/engine_main.cpp
    src/engine/GstCameraCapture.cpp
    src/engine/InferenceEngine.cpp
    src/engine/VisionEngine.cpp
    src/engine/preprocess.cpp
    src/engine/class_policy.cpp
    src/tracking/tracker.cpp
)

add_executable(farmeye_engine ${ENGINE_SOURCES})

target_include_directories(farmeye_engine PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${OpenCV_INCLUDE_DIRS}
)

target_link_libraries(farmeye_engine PRIVATE ${OpenCV_LIBS})

include(${CMAKE_CURRENT_SOURCE_DIR}/src/engine/Engine.cmake)
if(FARMEYE_ENABLE_GSTREAMER AND GST_FOUND)
    target_include_directories(farmeye_engine PRIVATE ${GST_INCLUDE_DIRS})
    target_link_directories(farmeye_engine PRIVATE ${GST_LIBRARY_DIRS})
    target_link_libraries(farmeye_engine PRIVATE ${GST_LIBRARIES})
    target_compile_definitions(farmeye_engine PRIVATE HAVE_GSTREAMER=1)
    message(STATUS "farmeye_engine: GStreamer ON")
else()
    message(WARNING "farmeye_engine: GStreamer OFF — capture stub")
endif()

# Model path helper (runtime cwd)
target_compile_definitions(farmeye_engine PRIVATE
    FARMEYE_SOURCE_DIR="${CMAKE_SOURCE_DIR}"
)

message(STATUS "OpenCV: ${OpenCV_VERSION}")
